### 1. WebSocket 信令服务

- [ ] 创建 `app/websocket/signaling.py`

  ```python
  from aiortc import RTCPeerConnection, RTCSessionDescription
  from fastapi import WebSocket
  from typing import Dict

  class SignalingConnection:
      def __init__(self, websocket: WebSocket):
          self.websocket = websocket
          self.peer_connection: RTCPeerConnection = None

  class SignalingHandler:
      def __init__(self):
          self.connections: Dict[str, SignalingConnection] = {}

      async def handle_connection(self, websocket: WebSocket, participant_id: str):
          connection = SignalingConnection(websocket)
          self.connections[participant_id] = connection

          try:
              while True:
                  message = await websocket.receive_json()
                  await self.handle_message(participant_id, message)
          except Exception as e:
              await self.handle_disconnect(participant_id)

      async def handle_message(self, participant_id: str, message: dict):
          connection = self.connections[participant_id]
          msg_type = message["type"]

          if msg_type == "offer":
              await self.handle_offer(connection, message["data"])
          elif msg_type == "answer":
              await self.handle_answer(connection, message["data"])
          elif msg_type == "ice-candidate":
              await self.handle_ice_candidate(connection, message["data"])

      async def handle_disconnect(self, participant_id: str):
          if participant_id in self.connections:
              connection = self.connections[participant_id]
              if connection.peer_connection:
                  await connection.peer_connection.close()
              del self.connections[participant_id]
  ```

- [ ] 创建 `app/services/connection_manager.py`

  ```python
  from aiortc import RTCConfiguration, RTCIceServer
  from fastapi import WebSocket
  from typing import Dict, Set

  class ConnectionManager:
      def __init__(self, stun_servers: List[str]):
          self.active_connections: Dict[str, Set[WebSocket]] = {}
          self.config = RTCConfiguration([
              RTCIceServer(urls=stun_servers)
          ])

      async def connect(self, websocket: WebSocket, room_id: str):
          await websocket.accept()
          if room_id not in self.active_connections:
              self.active_connections[room_id] = set()
          self.active_connections[room_id].add(websocket)

      async def disconnect(self, websocket: WebSocket, room_id: str):
          self.active_connections[room_id].remove(websocket)

      async def broadcast(self, message: dict, room_id: str, exclude: WebSocket = None):
          for connection in self.active_connections[room_id]:
              if connection != exclude:
                  await connection.send_json(message)
  ```

  ```python
  from aiortc.rtcicetransport import RTCIceCandidate

  class IceHandler:
      async def add_ice_candidate(self, pc: RTCPeerConnection, candidate: dict):
          candidate_obj = RTCIceCandidate(
              component=candidate["component"],
              foundation=candidate["foundation"],
              ip=candidate["ip"],
              port=candidate["port"],
              priority=candidate["priority"],
              protocol=candidate["protocol"],
              type=candidate["type"],
          )
          await pc.addIceCandidate(candidate_obj)

      async def on_ice_candidate(self, pc: RTCPeerConnection):
          @pc.on("icecandidate")
          async def on_ice_candidate(event):
              if event.candidate:
                  candidate_dict = {
                      "candidate": event.candidate.candidate,
                      "sdpMid": event.candidate.sdpMid,
                      "sdpMLineIndex": event.candidate.sdpMLineIndex,
                  }
                  return candidate_dict
  ```

  ```python
  from aiortc import RTCSessionDescription

  class SdpHandler:
      async def handle_offer(self, pc: RTCPeerConnection, offer: dict) -> RTCSessionDescription:
          offer_obj = RTCSessionDescription(sdp=offer["sdp"], type=offer["type"])
          await pc.setRemoteDescription(offer_obj)
          answer = await pc.createAnswer()
          await pc.setLocalDescription(answer)
          return answer

      async def handle_answer(self, pc: RTCPeerConnection, answer: dict):
          answer_obj = RTCSessionDescription(sdp=answer["sdp"], type=answer["type"])
          await pc.setRemoteDescription(answer_obj)
  ```

  ```python
  class AudioHandler:
      def __init__(self):
          self.audio_tracks = {}  # {participant_id: MediaStreamTrack}

      async def handle_track(self, track: MediaStreamTrack, participant_id: str):

          if participant_id in self.audio_tracks:
  ```

  ```python
  class RoomAudioManager:
      def __init__(self, connection_manager: ConnectionManager):
          self.rooms = {}  # {room_id: {participant_id: AudioParticipant}}
          self.connection_manager = connection_manager

      async def add_participant(self, room_id: str, participant: AudioParticipant):
          if room_id not in self.rooms:
              self.rooms[room_id] = {}
          self.rooms[room_id][participant.id] = participant
          await self._setup_peer_connections(room_id, participant)

      async def _setup_peer_connections(self, room_id: str, participant: AudioParticipant):
          for other_participant in self.rooms[room_id].values():
              if other_participant.id != participant.id:
                  # 建立双向连接
                  pc1 = await self.connection_manager.create_peer_connection()
                  pc2 = await self.connection_manager.create_peer_connection()
                  participant.peer_connections[other_participant.id] = pc1
                  other_participant.peer_connections[participant.id] = pc2
  ```

## 启动服务

```python
# app/main.py
from fastapi import FastAPI, WebSocket
from app.websocket.signaling import SignalingHandler
from app.services.connection_manager import ConnectionManager

app = FastAPI()
signaling = SignalingHandler()
connection_manager = ConnectionManager(["stun:stun.l.google.com:19302"])

@app.websocket("/ws/{room_id}/{participant_id}")
async def websocket_endpoint(websocket: WebSocket, room_id: str, participant_id: str):
    await connection_manager.connect(websocket, room_id)
    try:
        await signaling.handle_connection(websocket, participant_id)
    finally:
        await connection_manager.disconnect(websocket, room_id)
```
